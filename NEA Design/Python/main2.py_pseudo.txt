SET clock TO pygame.time.Clock() # Creates a clock object that is used to track amount of time

from pygame.locals IMPORT * # Imports constants that are used by the Pygame module

pygame.init() # Initialises all imported Pygame modules 

pygame.display.set_caption("Carrom Game") # Sets program caption 

SET screen TO pygame.display.set_mode((500,700),0,32) # Sets the display resolution



SET font TO pygame.font.SysFont('Arial Bold', 40) # Sets the font used within the program



DEFINE FUNCTION draw_text(text, font, color, surface, x, y): # Writes text onto the screen of the program

    SET textobj TO font.render(text, 1, color)

    SET textrect TO textobj.get_rect()

    SET textrect.topleft TO (x,y)

    surface.blit(textobj, textrect)

DEFINE FUNCTION main_menu(): 



    WHILE True:

        

        screen.fill(BLACK) # Fills the screen IN the desired colour

        draw_text("Main Menu", font, WHITE, screen, 20, 20)



        SET mx, my TO pygame.mouse.get_pos() # Retrieves the current position of where the mouse is



        SET button_1 TO pygame.Rect(50,100,200,50)

        SET button_2 TO pygame.Rect(50,200,200,50)

        SET button_3 TO pygame.Rect(50,300,200,50)

        SET button_4 TO pygame.Rect(50,400,200,50)

        SET button_5 TO pygame.Rect(50,500,200,50)

        SET button_6 TO pygame.Rect(50,600,200,50)



        IF button_1.collidepoint((mx,my)): # Checks IF mouse location overlaps button 1 location

            IF click: # Checks IF button has been clicked

                game()

        IF button_2.collidepoint((mx, my)):

            IF click:

                options()

        IF button_3.collidepoint((mx, my)):

            IF click:

                gameMode1()

        IF button_4.collidepoint((mx, my)): 

            IF click:

                gameMode2()

        IF button_5.collidepoint((mx, my)): 

            IF click:

                settings()

        IF button_6.collidepoint((mx, my)): 

            IF click:

                quit_game()

        

        pygame.draw.rect(screen, WHITE, button_1) # Draws a rectangle onto the screen IN white colour FOR button 1

        draw_text("Main Game", font, BLACK, screen, 70, 110) # Writes text onto the screen with the specified font at the start with a black colour onto the screen

        

        pygame.draw.rect(screen, WHITE, button_2)

        draw_text("Options", font, BLACK, screen, 90, 210)

        

        pygame.draw.rect(screen, WHITE, button_3)

        draw_text("Game Mode 1", font, BLACK, screen, 60, 310)

        

        pygame.draw.rect(screen, WHITE, button_4)

        draw_text("Game Mode 2", font, BLACK, screen, 60, 410)



        pygame.draw.rect(screen, WHITE, button_5)

        draw_text("Settings", font, BLACK, screen, 90, 510)



        pygame.draw.rect(screen, WHITE, button_6)

        draw_text("Quit", font, BLACK, screen, 120, 610)



        SET click TO False



        SET events TO pygame.event.get()

        FOR event IN events:

            IF event.type EQUALS QUIT: # Checks IF the cross (top right button) has been pressed

                pygame.quit() # Program ends

                sys.exit()

            IF event.type EQUALS KEYDOWN: # Checks IF a key has been pressed

                IF event.key EQUALS K_ESCAPE: # Checks IF escape key has been pressed

                    pygame.quit() # Programs ends

                    sys.exit()

            IF event.type EQUALS MOUSEBUTTONDOWN: # Checks IF mouse button has been pressed

                IF event.button EQUALS 1: # Checks IF button has been clicked

                    SET click TO True



        pygame.display.update() # Updates display

        clock.tick(FPS) #

            IF event.type EQUALS QUIT:

                pygame.quit()

            IF event.type EQUALS KEYDOWN:

                    SET run TO False

            IF event.type EQUALS KEYDOWN:

                IF event.key EQUALS K_q:

                    SET run TO False

                    pygame.quit()

        

        pygame.display.update()

        clock.tick(FPS)
